+#if UNITY_ANDROID

+public class NotificationManager : MonoBehaviour

+    public static NotificationManager Instance { get; private set; }
+    public bool NotificationsEnabled { get; private set; } = false; // Default to false
+    private const string AndroidChannelId = "default_channel";

+    private void Awake()
+    {
+        if (Instance == null)
+        {
+            Instance = this;
+            DontDestroyOnLoad(gameObject);
+            // Load the notificationsEnabled flag from PlayerPrefs
+            NotificationsEnabled = PlayerPrefs.GetInt(NotificationsEnabledKey, 0) == 1;
+        }
+        else
+        {
+            Destroy(gameObject);
+        }
+    }
+
+    private void Start()
+    {
+        Init();
+    }
+
+    void OnEnable()
+    {
+#if UNITY_ANDROID
+        AndroidNotificationCenter.OnNotificationReceived += OnNotificationReceived;
+#elif UNITY_IOS
+        iOSNotificationCenter.OnNotificationReceived += OnNotificationReceived;
+#endif
+    }
+
+    void OnDisable()
+    {
+#if UNITY_ANDROID
+        AndroidNotificationCenter.OnNotificationReceived -= OnNotificationReceived;
+#elif UNITY_IOS
+        iOSNotificationCenter.OnNotificationReceived -= OnNotificationReceived;
+#endif
+    }
+
+#if UNITY_ANDROID
+    private void OnNotificationReceived(AndroidNotificationIntentData data)
+    {
+    }
+#endif
+
+#if UNITY_IOS
+    private void OnNotificationReceived(iOSNotification notification)
+    {
+    }
+#endif
+
+    public void Init()
+    {
+        // Only initialize notifications if the user hasn't made a choice yet
+        InitializeNotifications();
+    }
+
+    private void OnApplicationFocus(bool focus)
+    {
+        if (focus)
+        {
+            // Check and update notification permission status
+#if UNITY_ANDROID
+            if (Permission.HasUserAuthorizedPermission("android.permission.POST_NOTIFICATIONS"))
+            {
+                PlayerPrefs.SetString(NotificationsPermission, PermissionGranted);
+            }
+            else
+            {
+                PlayerPrefs.SetString(NotificationsPermission, PermissionDenied);
+            }
+#elif UNITY_IOS
+            var notificationSettings = iOSNotificationCenter.GetNotificationSettings();
+            if (notificationSettings.AuthorizationStatus == AuthorizationStatus.Authorized)
+            {
+                PlayerPrefs.SetString(NotificationsPermission, PermissionGranted);
+            }
+            else
+            {
+                PlayerPrefs.SetString(NotificationsPermission, PermissionDenied);
+            }
+#endif
+            // Clear notifications and check for any triggered by app focus
+            ClearNotifications();
+            CheckIfAppOpenedFromNotification();
+        }
+        else
+        {
+            ClearNotifications();
+            ScheduleDominoNotifications();
+        }
+    }
+
+    public void EnableNotifications()
+    {
+        NotificationsEnabled = true;
+        PlayerPrefs.SetInt(NotificationsEnabledKey, 1);
+        PlayerPrefs.Save();
+
+        ClearNotifications();
+        ScheduleDominoNotifications();
+    }
+
+    public void DisableNotifications()
+    {
+        NotificationsEnabled = false;
+        PlayerPrefs.SetInt(NotificationsEnabledKey, 0);
+        PlayerPrefs.Save();
+        ClearNotifications();
+    }
+

+        // Lapsed Weekly reminder batch

+
+        // Schedule Weekly Leaderboard reminder batch
+        BatchScheduleWeeklyLeaderboardNotifications(WeeklyLeaderboardTitle, WeeklyLeaderboardMessage, 4, serverToday);
+

+        BatchSchedulePlagueRace(PlagueRaceTitle, PlagueRaceMessage, 5, serverToday);

+
+    private void ScheduleLivesRechargedNotification(string title, string message, DateTime serverToday)
+    {
+        if (LivesManager.IsInfiniteLivesActive)
+        {
+            return;
+        }
+
+        if (LivesManager.Instance != null && LivesManager.Instance.Lives < LivesManager.MaxLives)
+        {
+            var livesToRecharge = LivesManager.MaxLives - LivesManager.Instance.Lives;
+            var lastRewardTime = LivesManager.Instance.GetRegenerationTime();
+            TimeSpan timeSinceLastReward = DateTime.UtcNow - lastRewardTime;
+            float nextLifeTimeRemaining = (float)LivesManager.RegenTime.TotalSeconds - (float)timeSinceLastReward.TotalSeconds;
+            float allLivesTimeRemaining = nextLifeTimeRemaining + (livesToRecharge - 1) * (float)LivesManager.RegenTime.TotalSeconds;
+            if (allLivesTimeRemaining > 0)
+            { 
+                ScheduleNotification(title, message, DateTime.Now.AddSeconds(allLivesTimeRemaining));
+            }
+        }
+    }
+
+    private void ScheduleFinishFirstTutorialNotification(string title, string message, DateTime serverToday)
+    {
+        if (LevelManager.Instance != null && LevelManager.Instance.GetCurrentLevelNumber() < ZombieConstants.FirstTutorialFinished)
+        {
+            DateTime date = serverToday.AddHours(12);
+            ScheduleNotification(title, message, date);
+        }
+    }
+
+    private void DailyNotifications(string title, string message, DateTime serverToday)
+    {
+        DateTime date = serverToday.AddDays(1);
+        ScheduleNotification(title, message, date);
+    }
+
+    private void BatchScheduleWeeklyNotifications(string title, string message, int weeksToCover, DateTime referenceDate)
+    {
+        for (int i = 1; i <= weeksToCover; i++)
+        {
+            DateTime fireTime = referenceDate.AddDays(i * 7);
+            ScheduleNotification(title, message, fireTime);
+        }
+    }
+
+    private void BatchSchedulePlagueRace(string title, string message, int daysToCover, DateTime serverToday)
+    {
+        if (LevelManager.Instance != null && LevelManager.Instance.GetCurrentLevelNumber() >= ZombieConstants.PlagueRaceUnlockLevel)
+        {
+            for (int i = 1; i <= daysToCover; i++)
+            {
+                DateTime date = serverToday.AddDays(i);
+                DateTime fireTime = date.AddMinutes(15);
+                ScheduleNotification(title, message, fireTime);
+            }
+        }
+    }
+
